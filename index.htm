        function setupRealtimeSync() {

            const syncInd = document.getElementById('sync-indicator');

            const syncTxt = document.getElementById('sync-text');

 

            // 1. Sinhronizacija Vozača

            db.ref('drivers').on('value', (snapshot) => {

                const val = snapshot.val();

                drivers = val ? Object.values(val) : getInitialDrivers();

                populateDriverDropdowns();

                renderFleetAndDrivers();

            });

 

            // 2. Sinhronizacija Voznog Parka

            db.ref('fleet').on('value', (snapshot) => {

                const val = snapshot.val();

                fleet = val ? Object.values(val) : getInitialFleet();

                renderFleetAndDrivers();

            });

 

            // 3. Sinhronizacija Rezervacija

            db.ref('reservations').on('value', (snapshot) => {

                const val = snapshot.val();

                if (val) {

                    reservations = Object.values(val).sort((a,b) => new Date(b.dateTime) - new Date(a.dateTime));

                } else {

                    reservations = getInitialReservations();

                }

                

                syncInd.className = "w-2 h-2 rounded-full bg-emerald-500";

                syncTxt.innerText = "Sinhronizovano";

 

                updateDashboard();

                renderReservationsTable();

            }, (error) => {

                syncInd.className = "w-2 h-2 rounded-full bg-red-500";

                syncTxt.innerText = "Greška";

            });

        function exportToCSV() {

            let csv = 'ID,Ime Putnika,Telefon,Tip,Datum,Polaziste,Odrediste,Cena,Status\n';

            link.href = URL.createObjectURL(blob);

            link.setAttribute("download", `transferi_izvoz_${new Date().toISOString().slice(0,10)}.csv`);

            document.body.appendChild(link);

            link.click();

            document.body.removeChild(link);

        }

        function getStatusBadge(status) {

            switch (status) {

                case 'Potvrđeno':

                    return `<span class="bg-emerald-500/10 text-emerald-400 border border-emerald-500/20 text-xs px-2.5 py-1 rounded-full font-medium">Potvrđeno</span>`;

                case 'Na čekanju':

                    return `<span class="bg-amber-500/10 text-amber-400 border border-amber-500/20 text-xs px-2.5 py-1 rounded-full font-medium">Na čekanju</span>`;

                case 'Završeno':

                    return `<span class="bg-blue-500/10 text-blue-400 border border-blue-500/20 text-xs px-2.5 py-1 rounded-full font-medium">Završeno</span>`;

                case 'Otkazano':

                    return `<span class="bg-rose-500/10 text-rose-400 border border-rose-500/20 text-xs px-2.5 py-1 rounded-full font-medium">Otkazano</span>`;

                default:

                    return `<span class="bg-neutral-800 text-neutral-400 text-xs px-2.5 py-1 rounded-full">${status}</span>`;

            }

        }

 

        function formatDateTime(dtStr) {

            if (!dtStr) return '-';

            const d = new Date(dtStr);

            return d.toLocaleDateString('sr-RS', { day: '2-digit', month: '2-digit', year: 'numeric' }) + ' u ' + d.toLocaleTimeString('sr-RS', { hour: '2-digit', minute: '2-digit' });

        }
